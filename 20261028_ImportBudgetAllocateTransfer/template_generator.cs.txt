DataTable Q(string sql)
{
    using var cmd = new OracleCommand(sql, con);
    var dt = new DataTable();
    using var r = cmd.ExecuteReader();
    dt.Load(r);
    return dt;
}

string MasterName(Master m, string[] p)
{
    var id = p[m.SegNo - 1];
    var rows = m.Data.AsEnumerable().Where(r => S(r[0]) == id).ToList();
    if (m.PairWithAccount) { var pr = rows.FirstOrDefault(r => S(r[2]) == p[9]); if (pr != null) return S(pr[1]); }
    return rows.Count > 0 ? S(rows[0][1]) : "";
}

            // หาด้วยคู่ บัญชี.บัญชีย่อย ก่อน — ไม่เจอค่อยหาด้วยรหัสบัญชีย่อยอย่างเดียว และบอกว่าผังอยู่ใต้บัญชีอื่น
            string acc = $"{L(ColSeg1 + accountSegIdx)}{r}";
            string byPair = $"INDEX({Rng(m, 2)},MATCH({acc}&\".\"&{seg},{Rng(m, m.PairCol)},0))&\"\"";
            lookup = $"IFERROR({byPair},IFERROR({byId}&\" (ใน master อยู่ใต้บัญชีอื่น)\",{notFound}))";
        }
        else lookup = $"IFERROR(IF({byId}=\"\",\"(ไม่มีชื่อใน master)\",{byId}),{notFound})";
        wsItem.Cell(r, ColSegName1 + s).FormulaA1 = $"IF({seg}=\"\",IF({segsBlank},\"\",\"-\"),{lookup})";
    }

static void Verify(string path)
{
    using var wb = new XLWorkbook(path);
    Console.WriteLine("sheets: " + string.Join(", ", wb.Worksheets.Select(w => w.Name)));
    foreach (var n in new[] { "TEMPLATE_VERSION", "TRANSFER_ROUNDNO", "TRANSFER_ID" })
        Console.WriteLine($"{n}: " + wb.DefinedNames.Single(x => x.Name == n).Ranges.First().FirstCell().GetString());
    var item = wb.Worksheet("รายการโอน");
    var head = wb.Worksheet("ข้อมูลใบโอน");
    var acc = wb.Worksheet("ข้อมูลบัญชี");
    var cc = wb.Worksheet("ศูนย์ต้นทุน");
    var bank = wb.Worksheet("บัญชีธนาคาร");
    var bkRef = wb.Worksheet("ข้อมูลบัญชีธนาคาร");
    Console.WriteLine("item headers: " + string.Join(" | ", Enumerable.Range(1, 25).Select(c => $"{XLHelper.GetColumnLetterFromNumber(c)}:{item.Cell(4, c).GetString()}")));
    foreach (var s in new[] { "แหล่งเงิน", "แผนงาน", "ผลผลิต", "กิจกรรม", "ประเภทรายจ่าย", "รหัสงบประมาณ", "รหัสบัญชี", "รหัสบัญชีย่อย", "หน่วยงาน", "ศูนย์ต้นทุน" })
        Console.WriteLine($"  sheet {s}: {wb.Worksheet(s).Range("A5:A6000").CellsUsed().Count()} rows");

    var ccRows = cc.Range("A5:E2000").Rows().Where(x => x.Cell(1).GetString() != "").ToList();
    var cc1 = ccRows[0];
    var cc2 = ccRows.First(x => x.Cell(5).GetString() != cc1.Cell(5).GetString());
    const int cDept = 10, cCc = 11, cAmt = 12;
    void Paste(int row, int accRow) { for (int s = 0; s < 8; s++) item.Cell(row, 2 + s).Value = acc.Cell(accRow, 1 + s).GetString(); }
    void Org(int row, IXLRangeRow ccRow, string deptOverride = null) { item.Cell(row, cDept).Value = deptOverride ?? ccRow.Cell(5).GetString(); item.Cell(row, cCc).Value = ccRow.Cell(1).GetString(); }
    int FindAcc(Func<IXLRangeRow, bool> f) => acc.Range("A5:H2000").Rows().First(x => x.Cell(1).GetString() != "" && f(x)).RowNumber();
    int blankPlanRow = FindAcc(x => x.Cell(2).GetString() == "");
    int otherParentRow = FindAcc(x => x.Cell(7).GetString() == "5102010199" && x.Cell(8).GetString() == "230041");

    Paste(5, 5); Org(5, cc1); item.Cell(5, cAmt).Value = 150000;                                    // ถูกต้อง
    Paste(6, 6); Org(6, cc2); item.Cell(6, cAmt).Value = 87500.5;                                   // ถูกต้อง
    Paste(7, 5); item.Cell(7, 9).Value = "999999"; item.Cell(7, cDept).Value = "9999999999"; item.Cell(7, cCc).Value = "1234567890"; item.Cell(7, cAmt).Value = 10; // บัญชีย่อยไม่มี + หน่วยงาน/ศูนย์ไม่มี
    Paste(8, 5); Org(8, cc2, cc1.Cell(5).GetString()); item.Cell(8, cAmt).Value = 1;              // ศูนย์ต้นทุนสังกัดหน่วยงานอื่น
    Paste(9, 5); item.Cell(9, 2).Value = double.Parse(acc.Cell(5, 1).GetString()); item.Cell(9, 3).Value = double.Parse(acc.Cell(5, 2).GetString());
    item.Cell(9, cDept).Value = double.Parse(cc1.Cell(5).GetString()); item.Cell(9, cCc).Value = double.Parse(cc1.Cell(1).GetString()); item.Cell(9, cAmt).Value = 1; // วางเป็นตัวเลข
    Paste(10, 5); item.Cell(10, 7).Value = double.Parse(acc.Cell(5, 6).GetString()); Org(10, cc1); item.Cell(10, cAmt).Value = 1; // รหัสงบ 20 หลักเป็นตัวเลข
    Paste(11, 5); item.Cell(11, 3).Value = "55555"; Org(11, cc1); item.Cell(11, cAmt).Value = 1;   // แผนงานไม่มี
    Paste(12, blankPlanRow); Org(12, cc1); item.Cell(12, cAmt).Value = 1;                           // แผนงานว่าง
    Paste(13, 5); item.Cell(13, 3).Value = "00000"; Org(13, cc1); item.Cell(13, cAmt).Value = 1;    // แผนงานมีใน master แต่ชุดไม่มี
    Paste(14, otherParentRow); Org(14, cc1); item.Cell(14, cAmt).Value = 1;                          // บัญชีย่อยใน master อยู่ใต้บัญชีอื่น

    for (int i = 5; i <= 15; i++)
        Console.WriteLine($"row {i}: A=[{item.Cell(i, 1).Value}] N=[{item.Cell(i, 14).Value}] O=[{item.Cell(i, 15).Value}] P=[{item.Cell(i, 16).Value}] Q=[{item.Cell(i, 17).Value}]\n" +
            "        " + string.Join(" | ", Enumerable.Range(18, 8).Select(c => item.Cell(i, c).GetFormattedString())));
    Console.WriteLine($"head count=[{head.Cell(14, 2).Value}] sum=[{head.Cell(15, 2).Value}]");

    var acc1 = bkRef.Cell(5, 1).GetString(); var acc2 = bkRef.Cell(6, 1).GetString();
    bank.Cell(5, 1).Value = cc.Cell(5, 1).GetString(); bank.Cell(5, 3).Value = acc1; bank.Cell(5, 4).Value = acc2;
    bank.Cell(6, 1).Value = cc.Cell(5, 1).GetString(); bank.Cell(6, 3).Value = acc1; bank.Cell(6, 4).Value = acc1;
    for (int i = 5; i <= 6; i++) Console.WriteLine($"bank {i}: B=[{bank.Cell(i, 2).Value}] E=[{bank.Cell(i, 5).Value}]");

    int formulas = 0, errors = 0;
    foreach (var ws in wb.Worksheets)
        foreach (var c in ws.CellsUsed(XLCellsUsedOptions.All).Where(c => c.HasFormula))
        {
            formulas++;
            try { if (c.Value.IsError) { errors++; if (errors <= 5) Console.WriteLine($"ERR {ws.Name}!{c.Address}: {c.Value}"); } }
            catch (Exception e) { errors++; if (errors <= 5) Console.WriteLine($"EXC {ws.Name}!{c.Address}: {e.Message}"); }
        }
    Console.WriteLine($"formulas={formulas} errors={errors}");
    Console.WriteLine($"B5 locked={item.Cell(5, 2).Style.Protection.Locked} N5 locked={item.Cell(5, 14).Style.Protection.Locked} protected={item.IsProtected}");
}
